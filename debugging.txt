# Rejoin Engine - diagnostic commands for Termux
# Buka 1 clone delta (misal com.apengjers.v3), lalu jalankan perintah-perintah di bawah
# SATU PER SATU di Termux, dan kirim hasil outputnya ke sini.

# 1) Perintah TERPENTING: cek process utama clone (muncul angka = process persis ber-nama ini)
pidof com.apengjers.v3

# 2) Cek process yang tersisa dengan nama apengjers
pgrep -af apengjers

# 3) Daftar process yang mengandung "roblox"
ps -A | grep -i roblox

# 4) Daftar semua process yang mengandung "apengjers"
ps -A | grep -i apengjers

# 5) Kalau "ps -A" error "garbage option", gunakan ini:
ps | grep -i apengjers
